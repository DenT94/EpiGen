% ---------------------------------------------------------------------------
\begin{frame}{The problem}
  \begin{itemize}
    \item Broad-spectrum antibiotic proteins kill \textbf{everything} --- pathogens
      \emph{and} commensal (``good'') bacteria.
    \item Killing commensals has real downstream cost: dysbiosis, infection
      susceptibility, long antibiotic-course side effects.
    \item We want a drug that's still fully active against pathogens, but
      \textbf{locally silenced} wherever a healthy microbiome already is.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{The idea: a protease-gated switch}
  \begin{itemize}
    \item Commensal \emph{Lactobacillus} spp.\ secrete a protease (lactocepin,
      PrtP) that pathogens don't.
    \item Engineer lactocepin's substrate motif into a surface-exposed loop of
      the antibiotic protein.
    \item Near commensals: local protease cleaves the motif, inactivates the
      drug. Everywhere else: drug stays fully active.
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{The catch}
    Inserting a foreign motif destabilizes the scaffold. We need
    \textbf{compensatory mutations} that restore structural stability
    \emph{without} burying the motif (it has to stay cleavable).
  \end{block}
\end{frame}

% ---------------------------------------------------------------------------

\begin{frame}{Pipeline: fold $\to$ invert $\to$ refold $\to$ search}
  \begin{enumerate}
    \item Fold the edited sequence (ESMFold2), gate on pLDDT/PAE.
    \item Inverse-fold to propose compensatory candidates (ProteinMPNN),
      MCMC search over the whole scaffold minus the fixed edit.
    \item Score every candidate with three independent experts: \textbf{ESM2}
      (sequence LM), \textbf{ProteinMPNN} (structure-conditioned),
      \textbf{Evo2} (DNA-level, causal) --- combined into one MCMC objective.
    \item Refold top candidates, self-consistency (TM-score) gate before
      trusting anything downstream.
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Three legs of evidence, per candidate}
  \begin{columns}
    \begin{column}{0.33\textwidth}
      \centering\textbf{Contact/energy}\\[0.3em]
      \small Per-neighbor $\Delta$distance, $\Delta$contact energy,
      $\Delta$pLDDT/PAE near every changed residue.
    \end{column}
    \begin{column}{0.33\textwidth}
      \centering\textbf{SAE interpretability}\\[0.3em]
      \small ESM-C sparse-autoencoder feature deltas across
      WT / edit-only / compensated --- top-$k$ surfaced, not a full sweep.
    \end{column}
    \begin{column}{0.33\textwidth}
      \centering\textbf{Agent explanation}\\[0.3em]
      \small Grounded in the numeric deltas above --- flags itself if a
      claimed rescue contradicts the numbers.
    \end{column}
  \end{columns}
  \vspace{1em}
  \centering\small
  All three feed one plain-language explanation of \emph{whether and how}
  a compensatory mutation rescues stability.
\end{frame}
